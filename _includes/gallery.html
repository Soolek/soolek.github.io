{%- comment -%} Folder-driven photo gallery with a same-page lightbox (GLightbox, see _layouts/default.html). Usage: {% include gallery.html dir="/assets/images/driver/e46/" name="e46" alt="BMW E46 drift car" %} - every .jpg/.jpeg/.png in `dir` becomes a tile; nothing is hard-coded in the page - thumbnails `_t.` are generated by scripts/make_thumbnails.py (a photo without one shows full size) - captions / alt texts come from _data/galleries.yml: -> -> - the page must set `gallery: true` in its front matter so the lightbox CSS/JS are loaded {%- endcomment -%} {%- assign gallery_dir = include.dir -%} {%- assign gallery_lang = page.lang | default: site.lang -%} {%- assign captions = site.data.galleries[include.name] -%} {%- assign photos = site.static_files | where_exp: "f", "f.path contains gallery_dir" | sort: "path" -%} {%- assign n = 0 -%}
{%- for photo in photos -%} {%- unless photo.extname == ".jpg" or photo.extname == ".jpeg" or photo.extname == ".png" -%}{%- continue -%}{%- endunless -%} {%- assign thumb_suffix = "_t" | append: photo.extname -%} {%- if photo.name contains thumb_suffix -%}{%- continue -%}{%- endif -%} {%- assign n = n | plus: 1 -%} {%- assign thumb_path = photo.path | replace: photo.extname, thumb_suffix -%} {%- assign thumb = site.static_files | where: "path", thumb_path | first -%} {%- assign caption = nil -%} {%- if captions and captions[photo.name] -%} {%- assign caption = captions[photo.name][gallery_lang] -%} {%- endif -%} {%- unless caption -%} {%- capture caption %}{{ include.alt | default: include.name }} - {% if gallery_lang == "pl" %}zdjęcie{% else %}photo{% endif %} {{ n }}{% endcapture -%} {%- endunless %} {%- endfor %}
